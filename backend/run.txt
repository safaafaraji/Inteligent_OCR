#!/usr/bin/env python3
import os
import sys
import subprocess
from pathlib import Path

def check_environment():
    """Vérifie l'environnement"""
    print("🔍 Vérification de l'environnement...")
    
    # Vérifier Tesseract
    try:
        tesseract_path = subprocess.check_output(["which", "tesseract"], text=True).strip()
        print(f"✅ Tesseract trouvé: {tesseract_path}")
        os.environ["TESSERACT_PATH"] = tesseract_path
    except subprocess.CalledProcessError:
        print("⚠️  Tesseract n'est pas installé")
        print("Installez-le avec:")
        print("  macOS: brew install tesseract poppler tesseract-lang")
        print("  Linux: sudo apt-get install tesseract-ocr tesseract-ocr-fra poppler-utils")
    
    # Vérifier et créer les répertoires
    directories = ["uploads", "static", "static/previews"]
    for directory in directories:
        Path(directory).mkdir(parents=True, exist_ok=True)
        print(f"✅ Répertoire: {directory}")
    
    # Vérifier le fichier .env
    env_file = ".env"
    if not Path(env_file).exists():
        print(f"📄 Création du fichier {env_file}...")
        with open(env_file, "w") as f:
            f.write("""# Application
DEBUG=True
HOST=0.0.0.0
PORT=8000

# Database
DATABASE_URL=sqlite:///./ocr_database.db

# JWT
SECRET_KEY=your-secret-key-change-this-in-production

# File upload
MAX_FILE_SIZE=10485760
UPLOAD_DIR=uploads
STATIC_DIR=static

# OCR Configuration
TESSERACT_PATH=/usr/local/bin/tesseract
DEFAULT_LANGUAGE=fra+eng
CONFIDENCE_THRESHOLD=0.6
""")
    
    return True

def main():
    """Fonction principale"""
    print("🚀 OCR Intelligent - Serveur de démarrage")
    print("=" * 50)
    
    check_environment()
    
    print("\n🌐 Démarrage du serveur...")
    print("📡 API: http://localhost:8000")
    print("📚 Documentation: http://localhost:8000/docs")
    print("👤 Interface: http://localhost:8080 (frontend)")
    print("🛑 CTRL+C pour arrêter")
    print("=" * 50)
    
    try:
        import uvicorn
        uvicorn.run(
            "app.main:app",
            host="0.0.0.0",
            port=8000,
            reload=True,
            log_level="info"
        )
    except KeyboardInterrupt:
        print("\n👋 Serveur arrêté")
    except Exception as e:
        print(f"❌ Erreur: {e}")
        sys.exit(1)

if __name__ == "__main__":
    main()
